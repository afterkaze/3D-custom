cmake_minimum_required(VERSION 3.16)
project(PyroCadeLeia LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_subdirectory(SR-lib)
add_executable(PyroCadeLeia leia-presenter.cpp)
target_compile_features(PyroCadeLeia PRIVATE cxx_std_17)
target_compile_definitions(PyroCadeLeia PRIVATE UNICODE _UNICODE NOMINMAX)
target_link_libraries(PyroCadeLeia PRIVATE SRLib::SR d3d11 dxgi d3dcompiler user32)
srlib_apply_delayload(PyroCadeLeia)
set_target_properties(PyroCadeLeia PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/bin")
